cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(flytec_carrier C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(flytec_carrier main.cpp)
target_include_directories(flytec_carrier PRIVATE ../source)
target_link_libraries(flytec_carrier pico_stdlib hardware_i2c pico_i2c_slave hardware_pwm hardware_clocks hardware_watchdog hardware_sync)
pico_enable_stdio_usb(flytec_carrier 0)
pico_enable_stdio_uart(flytec_carrier 0)
pico_add_extra_outputs(flytec_carrier)
